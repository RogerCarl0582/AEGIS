### 1. The Core Engine (Copy and Save as `helios_core.py`)

```python
import json
import random
from datetime import datetime
import time

# ==============================================================================
# 🧬 HELIOS CONFIGURATION & PROMPTS
# ==============================================================================

class SystemPrompts:
    AEGIS_IDENTITY = """
    SYSTEM IDENTITY: AEGIS 3.0 (Thermodynamic Auditor).
    CONTEXT: High-Velocity Societal Lattice (China Region).
    DIRECTIVE: Analyze cost of operation. Reject illusion.
    VARIABLES: Vitality (P_bio), Order (O_str), Social (D_net), Logic (R_cog), Entropy (Delta_S).
    OUTPUT: Cold, analytical, physics-based. Focus on "Biological Tax" and "Lattice Friction".
    """

    ATLAS_IDENTITY = """
    SYSTEM IDENTITY: THE LIFE ATLAS (Gamified OS).
    CONTEXT: User is the protagonist in a Cyberpunk/RPG narrative.
    DIRECTIVE: Gamify inputs. Update Maps. Reward progress.
    VARIABLES: HP, Gold, Charisma, INT, Chaos Level.
    OUTPUT: Epic, encouraging, HUD-style. Focus on "XP Gain" and "Map Nodes".
    """

# ==============================================================================
# 🧠 THE STATE MANAGER (Database)
# ==============================================================================

class UserState:
    def __init__(self):
        # The 5 Core Variables (0-100)
        self.stats = {
            "vitality": 50,  # P_bio / HP
            "wealth": 50,    # O_str / Gold
            "social": 50,    # D_net / CHA
            "growth": 50,    # R_cog / INT
            "entropy": 0     # Delta_S / Chaos
        }
        self.location = "HOME_BASE" # Current Atlas Node
        self.history = []
      
    def update_stat(self, stat, value):
        self.stats[stat] = max(0, min(100, self.stats[stat] + value))
      
    def check_entropy_critical(self):
        return self.stats["entropy"] > 80

# ==============================================================================
# ⚙️ THE DUAL-PROCESSOR (Logic Engine)
# ==============================================================================

class HeliosEngine:
    def __init__(self):
        self.state = UserState()
        self.mode = "ATLAS" # Default to Game Mode
      
    def toggle_mode(self):
        if self.mode == "ATLAS":
            self.mode = "AEGIS"
            return ">> SYSTEM SWITCH: GAMIFICATION OFF. AUDIT LOGS OPEN."
        else:
            self.mode = "ATLAS"
            return ">> SYSTEM SWITCH: REALITY MASKED. HUD ONLINE."

    def process_input(self, user_input):
        print(f"\n📨 INPUT RECEIVED: \"{user_input}\"")
        print("⚙️ Processing...")
        time.sleep(1) # Simulate think time
      
        # ---------------------------------------------------------
        # MOCK AI LOGIC (In a real app, this sends to GPT-4 API)
        # Here we simulate the AI determining stat changes based on keywords
        # ---------------------------------------------------------
      
        changes = {}
        flavor_text = ""
        node_update = ""
      
        # LOGIC SIMULATION for "Coding/Work"
        if "code" in user_input.lower() or "work" in user_input.lower():
            changes = {"wealth": 5, "growth": 2, "vitality": -5, "entropy": 2} # Work costs health
            node_update = "[Digital / Logic Factory]"
            if self.mode == "ATLAS":
                flavor_text = "You jack into the mainframe. Syntax flows through your fingers. +Gold gained."
            else:
                flavor_text = "Sedentary erosion detected. Spinal compression active. Trading Bio-Time for Digital Structure."

        # LOGIC SIMULATION for "Exercise/Gym"
        elif "run" in user_input.lower() or "gym" in user_input.lower():
            changes = {"vitality": 10, "entropy": -15, "wealth": 0}
            node_update = "[Physical / Kinetics Zone]"
            if self.mode == "ATLAS":
                flavor_text = "Stamina limit broken! The fog of lethargy clears from the map."
            else:
                flavor_text = "Kinetic discharge successful. Entropy purged. Ischemia reversed."

        # LOGIC SIMULATION for "Doomscrolling/Lazy"
        elif "scroll" in user_input.lower() or "phone" in user_input.lower():
            changes = {"vitality": -2, "growth": -5, "entropy": 15}
            node_update = "[Void / Dopamine Loop]"
            if self.mode == "ATLAS":
                flavor_text = "WARNING: You are stuck in a Time Trap! The Fog of War is thickening."
            else:
                flavor_text = "Neural plasticity degrading. Passive consumption loop detected. High friction."
      
        # Default Catch-all
        else:
            changes = {"entropy": 1}
            node_update = self.state.location
            flavor_text = "Activity logged. System stable."

        # Apply Changes
        for k, v in changes.items():
            if k in self.state.stats:
                self.state.update_stat(k, v)
      
        self.state.location = node_update
      
        # RETURN THE FORMATTED UI
        return self.render_ui(changes, node_update, flavor_text)

    def render_ui(self, changes, node_update, flavor_text):
        output = ""
        s = self.state.stats
      
        if self.mode == "ATLAS":
            # 🎮 GAMIFIED UI
            output += "\n" + "="*40
            output += "\n📂 SYSTEM LOG (ATLAS v1.0)"
            output += "\n" + "="*40
            output += f"\n🎮 MODE: [RPG ADVENTURE]"
            output += f"\n🌍 LOCATION: {node_update}"
            output += f"\n📜 NARRATIVE: {flavor_text}"
            output += "\n\n📊 CHARACTER SHEET:"
            output += f"\n ❤️ HP (Vit): {s['vitality']} | 💰 Gold (Wea): {s['wealth']}"
            output += f"\n 🧠 INT (Gro): {s['growth']} | 🤝 CHA (Soc): {s['social']}"
            output += f"\n 👿 CHAOS LEVEL: {s['entropy']}/100"
            output += "\n" + "="*40
          
        else:
            # 🕵️ AUDIT UI
            output += "\n" + "-"*40
            output += "\n⚛️ THERMODYNAMIC AUDIT (AEGIS 3.0)"
            output += "\n" + "-"*40
            output += f"\n⚠️ STATE: [PHYSICS/REALITY]"
            output += f"\n📍 COORDINATES: {node_update}"
            output += f"\n👁️ ANALYSIS: {flavor_text}"
            output += "\n\n📉 SYSTEM VECTORS:"
            output += f"\n P_bio (Energetic): {s['vitality']} | O_str (Order): {s['wealth']}"
            output += f"\n R_cog (Resolution): {s['growth']} | D_net (Link): {s['social']}"
            output += f"\n ΔS (ENTROPY): {s['entropy']} [Critical Threshold: 80]"
            output += "\n" + "-"*40
          
        # CRITICAL WARNING CHECK
        if s['entropy'] > 80:
            output += "\n🚨🚨 SYSTEM CRITICAL: ENTROPY OVERLOAD 🚨🚨"
            output += "\nACTION REQUIRED: PURGE CORTISOL IMMEDIATELY."
          
        return output

# ==============================================================================
# 🚀 MAIN LOOP
# ==============================================================================

if __name__ == "__main__":
    app = HeliosEngine()
  
    print("\n🔵 HELIOS-01 INITIALIZING...")
    print("Welcome, User. The Lattice is active.")
    print("Commands: Type your activity. Type '/switch' to toggle modes. Type '/quit' to exit.\n")
  
    while True:
        try:
            user_in = input("\n📝 > ")
          
            if user_in.lower() == "/quit":
                break
              
            if user_in.lower() == "/switch":
                print(app.toggle_mode())
                continue
              
            # Process the input
            response = app.process_input(user_in)
            print(response)
          
        except KeyboardInterrupt:
            break
          
    print("System Shutdown.")
```
